cmake_minimum_required(VERSION 3.16)
project(PDALab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PDA_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(PDA_STRICT_WARNINGS "Also diagnose conversions and shadowing in legacy algorithms" OFF)
option(PDA_FULL_TESTS "Register long-running official integrations (requires fetched resources)" OFF)
add_library(pda_options INTERFACE)
target_include_directories(pda_options INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pda_options INTERFACE -Wall -Wextra -Wpedantic)
  if(PDA_STRICT_WARNINGS)
    target_compile_options(pda_options INTERFACE -Wconversion -Wshadow)
  endif()
  if(PDA_SANITIZERS)
    target_compile_options(pda_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pda_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()
find_package(Boost REQUIRED)
add_library(lab1_core Lab01/src/Block_Creating.cpp Lab01/src/Point_Finding.cpp
  Lab01/src/Neighbor_Finding.cpp)
target_include_directories(lab1_core PUBLIC Lab01/header)
target_link_libraries(lab1_core PUBLIC pda_options)
add_executable(Lab1 Lab01/src/main.cpp)
target_link_libraries(Lab1 PRIVATE lab1_core)
add_library(lab2_core Lab02/src/parser.cpp Lab02/src/packing.cpp Lab02/src/tree.cpp
  Lab02/src/annealer.cpp Lab02/src/report.cpp)
target_include_directories(lab2_core PUBLIC Lab02/include)
target_link_libraries(lab2_core PUBLIC pda_options)
add_executable(Lab2 Lab02/main.cpp)
target_link_libraries(Lab2 PRIVATE lab2_core)
add_executable(Legalizer Lab03/main.cpp Lab03/parser.cpp Lab03/legalizer.cpp)
target_include_directories(Legalizer PRIVATE Lab03/include)
target_include_directories(Legalizer SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
add_executable(D2DGRter Lab04/main.cpp Lab04/parser.cpp Lab04/layered_router.cpp)
target_include_directories(D2DGRter PRIVATE Lab04/inc)
foreach(pair "Lab1;Lab01" "Lab2;Lab02" "Legalizer;Lab03" "D2DGRter;Lab04")
  list(GET pair 0 target)
  list(GET pair 1 lab)
  target_link_libraries(${target} PRIVATE pda_options)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/${lab})
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(lab1_core_test tests/lab1_core.cpp)
  target_link_libraries(lab1_core_test PRIVATE lab1_core)
  add_test(NAME lab1_core COMMAND lab1_core_test)
  add_executable(lab2_core_test tests/lab2_core.cpp)
  target_link_libraries(lab2_core_test PRIVATE lab2_core)
  add_test(NAME lab2_core COMMAND lab2_core_test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME lab4_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/lab4_oracle.py
    --bin-root ${CMAKE_BINARY_DIR})
  set_tests_properties(lab4_oracle PROPERTIES TIMEOUT 120)
  add_test(NAME regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/regression.py
    --bin-root ${CMAKE_BINARY_DIR})
  set_tests_properties(regression PROPERTIES TIMEOUT 120)
  if(PDA_FULL_TESTS)
    foreach(lab Lab01 Lab02 Lab03 Lab04)
      add_test(NAME official_${lab} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/run.py
        --lab ${lab} --bin-root ${CMAKE_BINARY_DIR})
      set_tests_properties(official_${lab} PROPERTIES LABELS official TIMEOUT 10800 RUN_SERIAL TRUE)
    endforeach()
  endif()
endif()
