cmake_minimum_required(VERSION 3.16)
project(PDALab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PDA_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(PDA_STRICT_WARNINGS "Also diagnose conversions and shadowing" OFF)
option(PDA_WARNINGS_AS_ERRORS "Treat project compiler warnings as errors" OFF)
option(PDA_FULL_TESTS "Register long-running official integrations (requires fetched resources)" OFF)
option(PDA_THREAD_SANITIZER "Enable a separate ThreadSanitizer build" OFF)
option(PDA_FUZZING "Build Clang libFuzzer parser targets with ASan/UBSan" OFF)
if(PDA_THREAD_SANITIZER AND (PDA_SANITIZERS OR PDA_FUZZING))
  message(FATAL_ERROR "ThreadSanitizer requires a separate build from ASan/fuzzing")
endif()
if(PDA_FUZZING AND NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "PDA_FUZZING requires Clang with libFuzzer")
endif()
add_library(pda_options INTERFACE)
target_include_directories(pda_options INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pda_options INTERFACE -Wall -Wextra -Wpedantic)
  if(PDA_STRICT_WARNINGS)
    target_compile_options(pda_options INTERFACE -Wconversion -Wshadow)
  endif()
  if(PDA_WARNINGS_AS_ERRORS)
    target_compile_options(pda_options INTERFACE -Werror)
  endif()
  if(PDA_SANITIZERS OR PDA_FUZZING)
    target_compile_options(pda_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pda_options INTERFACE -fsanitize=address,undefined)
  endif()
  if(PDA_THREAD_SANITIZER)
    target_compile_options(pda_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(pda_options INTERFACE -fsanitize=thread)
  endif()
  if(PDA_FUZZING)
    target_compile_options(pda_options INTERFACE -fsanitize=fuzzer-no-link)
  endif()
endif()
find_package(Boost REQUIRED)
find_package(Threads REQUIRED)
add_library(lab1_core Lab01/src/TileList.cpp Lab01/src/Block_Creating.cpp Lab01/src/Point_Finding.cpp
  Lab01/src/Neighbor_Finding.cpp Lab01/src/parser.cpp Lab01/src/layout.cpp Lab01/src/report.cpp)
target_include_directories(lab1_core PUBLIC Lab01/header)
target_include_directories(lab1_core SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(lab1_core PUBLIC pda_options)
add_executable(Lab1 Lab01/src/main.cpp)
target_link_libraries(Lab1 PRIVATE lab1_core)
add_library(lab2_core Lab02/src/parser.cpp Lab02/src/packing.cpp Lab02/src/tree.cpp
  Lab02/src/annealer.cpp Lab02/src/parallel.cpp Lab02/src/undo.cpp Lab02/src/report.cpp)
target_include_directories(lab2_core PUBLIC Lab02/include)
target_link_libraries(lab2_core PUBLIC pda_options Threads::Threads)
add_executable(Lab2 Lab02/main.cpp)
target_link_libraries(Lab2 PRIVATE lab2_core)
add_library(lab3_core Lab03/parser.cpp Lab03/legalizer.cpp)
target_include_directories(lab3_core PUBLIC Lab03/include)
target_include_directories(lab3_core SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(lab3_core PUBLIC pda_options)
add_executable(Legalizer Lab03/main.cpp)
target_link_libraries(Legalizer PRIVATE lab3_core)
add_library(lab4_core Lab04/parser.cpp Lab04/legacy_router.cpp Lab04/layered_router.cpp Lab04/routing_state.cpp)
target_include_directories(lab4_core PUBLIC Lab04/inc)
target_link_libraries(lab4_core PUBLIC pda_options)
add_executable(D2DGRter Lab04/main.cpp)
target_link_libraries(D2DGRter PRIVATE lab4_core)
foreach(pair "Lab1;Lab01" "Lab2;Lab02" "Legalizer;Lab03" "D2DGRter;Lab04")
  list(GET pair 0 target)
  list(GET pair 1 lab)
  target_link_libraries(${target} PRIVATE pda_options)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/${lab})
endforeach()
include(CTest)
if(PDA_FUZZING)
  foreach(lab RANGE 1 4)
    add_executable(fuzz_lab${lab} tests/fuzz/parsers.cpp)
    target_compile_definitions(fuzz_lab${lab} PRIVATE PDA_FUZZ_LAB=${lab})
    target_link_libraries(fuzz_lab${lab} PRIVATE lab${lab}_core)
    target_link_options(fuzz_lab${lab} PRIVATE -fsanitize=fuzzer)
  endforeach()
endif()
if(BUILD_TESTING)
  add_executable(core_sessions_test tests/core_sessions.cpp)
  target_link_libraries(core_sessions_test PRIVATE lab3_core lab4_core)
  add_test(NAME core_sessions COMMAND core_sessions_test)
  add_executable(lab1_core_test tests/lab1_core.cpp)
  target_link_libraries(lab1_core_test PRIVATE lab1_core)
  add_test(NAME lab1_core COMMAND lab1_core_test)
  add_executable(lab2_core_test tests/lab2_core.cpp)
  target_link_libraries(lab2_core_test PRIVATE lab2_core)
  add_test(NAME lab2_core COMMAND lab2_core_test)
  add_executable(lab2_undo_test tests/lab2_undo.cpp)
  target_link_libraries(lab2_undo_test PRIVATE lab2_core)
  add_test(NAME lab2_undo COMMAND lab2_undo_test)
  add_executable(lab2_parallel_test tests/lab2_parallel.cpp)
  target_link_libraries(lab2_parallel_test PRIVATE lab2_core)
  add_test(NAME lab2_parallel COMMAND lab2_parallel_test)
  add_executable(lab2_random_test tests/lab2_random.cpp)
  target_link_libraries(lab2_random_test PRIVATE lab2_core Threads::Threads)
  add_test(NAME lab2_random COMMAND lab2_random_test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME lab4_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/lab4_oracle.py
    --bin-root ${CMAKE_BINARY_DIR})
  set_tests_properties(lab4_oracle PROPERTIES TIMEOUT 120)
  add_test(NAME regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/regression.py
    --bin-root ${CMAKE_BINARY_DIR})
  set_tests_properties(regression PROPERTIES TIMEOUT 120)
  add_test(NAME stress_smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/stress.py
    --bin-root ${CMAKE_BINARY_DIR} --scale smoke --output ${CMAKE_BINARY_DIR}/stress-smoke.json)
  set_tests_properties(stress_smoke PROPERTIES TIMEOUT 120)
  if(PDA_FULL_TESTS)
    foreach(lab Lab01 Lab02 Lab03 Lab04)
      add_test(NAME official_${lab} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/run.py
        --lab ${lab} --bin-root ${CMAKE_BINARY_DIR})
      set_tests_properties(official_${lab} PROPERTIES LABELS official TIMEOUT 10800 RUN_SERIAL TRUE)
    endforeach()
  endif()
endif()
